## Routers

Create new routers in `apps/web/src/server/api/routers`

```ts
// apps/web/src/server/api/routers/post.ts

import { z } from "zod";
import { createTRPCRouter, publicProcedure } from "@/server/api/trpc";
import { connect } from "@myapp/database";

const db = connect();

export const postRouter = createTRPCRouter({
  hello: publicProcedure
    .input(z.object({ text: z.string() }))
    .query(({ input }) => {
      return {
        greeting: `Hello ${input.text}`,
      };
    }),
});
```

Read from your database using `connect()`.

```ts
export const postRouter = createTRPCRouter({
  hello: publicProcedure
    .input(z.object({ text: z.string() }))
    .query(({ input }) => {
      return {
        greeting: `Hello ${input.text}`,
      };
    }),

  getLatest: publicProcedure.query(() => {
    return db.selectFrom("post").orderBy("createdAt", "desc").limit(1);
  }),
});
```

Update your database using `mutation`.

```ts
export const postRouter = createTRPCRouter({
  hello: publicProcedure
    .input(z.object({ text: z.string() }))
    .query(({ input }) => {
      return {
        greeting: `Hello ${input.text}`,
      };
    }),

  getLatest: publicProcedure.query(() => {
    return db.selectFrom("post").orderBy("createdAt", "desc").limit(1);
  }),

  create: publicProcedure
    .input(z.object({ content: z.string().min(1) }))
    .mutation(async ({ input }) => {
      return await db
        .insertInto("post")
        .values({
          createdAt: new Date(),
          content: input.content,
        })
        .returningAll();
    }),
});
```
