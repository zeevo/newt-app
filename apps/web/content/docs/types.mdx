title: "types"
description: "Add shareable types"
---

```bash
pnpm create newt-app -m types
```

This module adds a `types` package. Many projects find it useful to share types across
different `apps`. Defining types in a shared `package` allows you to have end-to-end
type safety.

```ts
// packages/types/src/query.ts
export interface User {
  id: string;
  name: string;
  email: string;
  role?: "admin" | "user";
}
```

Use your types in an `app` or `package`.

```ts
"use client";
import React from "react";
import type { User } from "@your-app/types/user";

export function UserCard({ user }: { user: User }) {
  return (
    <div className="rounded-xl border p-4 shadow-sm hover:shadow-md transition">
      <h2 className="text-lg font-semibold">{user.name}</h2>
      <p className="text-sm text-gray-500">{user.email}</p>
      {user.role && (
        <span className="mt-2 inline-block rounded bg-blue-100 px-2 py-1 text-xs font-medium text-blue-800">
          {user.role}
        </span>
      )}
    </div>
  );
}

```
